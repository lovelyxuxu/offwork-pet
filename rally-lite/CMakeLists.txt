cmake_minimum_required(VERSION 3.24)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")
endif()
project(RallyLite VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json
  URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)
add_library(rally_protocol STATIC src/protocol.cpp)
target_include_directories(rally_protocol PUBLIC src)
target_link_libraries(rally_protocol PUBLIC nlohmann_json::nlohmann_json)
add_library(rally_client STATIC src/client.cpp)
target_link_libraries(rally_client PUBLIC rally_protocol)
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601 WINVER=0x0601)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
  set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
  set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
  set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
  set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
  set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
  set(CURL_USE_LIBSSH OFF CACHE BOOL "" FORCE)
  set(CURL_USE_GSSAPI OFF CACHE BOOL "" FORCE)
  set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
  set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
  set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
  set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
  set(USE_LIBIDN2 OFF CACHE BOOL "" FORCE)
  set(HTTP_ONLY ON CACHE BOOL "" FORCE)
  foreach(feature LDAP LDAPS FTP FILE DICT TELNET TFTP RTSP POP3 IMAP SMTP GOPHER MQTT SMB)
    set(CURL_DISABLE_${feature} ON CACHE BOOL "" FORCE)
  endforeach()
  FetchContent_Declare(curl
    URL https://curl.se/download/curl-8.23.0.tar.xz
    URL_HASH SHA256=8dadd7d2536a314f982ad05a651b713f426bf7d02cc2c76cffb9796f2218e6c5
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(curl)
  target_sources(rally_client PRIVATE src/curl_transport.cpp src/win_storage.cpp)
  target_link_libraries(rally_client PUBLIC CURL::libcurl crypt32 shell32)
  add_executable(RallyLite WIN32 src/win_app.cpp)
  target_link_libraries(RallyLite PRIVATE rally_client gdiplus user32 gdi32)
  if(MSVC)
    target_compile_options(rally_protocol PRIVATE /utf-8)
    target_compile_options(rally_client PRIVATE /utf-8)
    target_compile_options(RallyLite PRIVATE /utf-8)
    target_link_options(RallyLite PRIVATE /SUBSYSTEM:WINDOWS,6.01)
  elseif(MINGW)
    target_link_options(RallyLite PRIVATE -static -Wl,--major-subsystem-version,6,--minor-subsystem-version,1)
  endif()
elseif(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  target_sources(rally_client PRIVATE src/mac_transport.mm src/mac_storage.mm)
  target_compile_options(rally_client PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(rally_client PUBLIC "-framework Foundation" "-framework Security")
  add_executable(RallyLite MACOSX_BUNDLE src/mac_app.mm)
  target_compile_options(RallyLite PRIVATE -fobjc-arc)
  target_link_libraries(RallyLite PRIVATE rally_client "-framework Cocoa" "-framework Carbon")
  set_target_properties(RallyLite PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
else()
  message(FATAL_ERROR "RallyLite supports Windows and macOS only")
endif()
enable_testing()
add_executable(rally_protocol_test tests/protocol_test.cpp)
target_link_libraries(rally_protocol_test PRIVATE rally_protocol)
add_test(NAME protocol COMMAND rally_protocol_test)
add_executable(rally_probe tests/client_probe.cpp)
target_link_libraries(rally_probe PRIVATE rally_client)
if(MSVC)
  target_compile_options(rally_protocol_test PRIVATE /utf-8)
  target_compile_options(rally_probe PRIVATE /utf-8)
endif()
install(TARGETS RallyLite BUNDLE DESTINATION . RUNTIME DESTINATION .)
install(FILES "${nlohmann_json_SOURCE_DIR}/LICENSE.MIT" DESTINATION licenses RENAME nlohmann-json.txt)
if(WIN32)
  install(FILES "${curl_SOURCE_DIR}/COPYING" DESTINATION licenses RENAME curl.txt)
endif()
